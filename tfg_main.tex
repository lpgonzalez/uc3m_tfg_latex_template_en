% --- PDF/A archival output (for the e-Archivo deposit) ---------------
% Change false to true to generate a PDF/A-2b file. For the final version, hide
% the demo chapter with \showDemoChapter=false (it is example content) and
% validate the PDF with veraPDF. This MUST stay at the very top:
% \DocumentMetadata has to be the first command, even before \documentclass, so
% this single switch lives here, not in tfg_vars.tex.
\def\pdfaOutput{false}
\def\tfgpdfayes{true}%
\ifx\pdfaOutput\tfgpdfayes
  \DocumentMetadata{pdfstandard=A-2b,pdfversion=1.7}% PDF/A-2b (PDF 1.7)
\else
  \pdfvariable minorversion=7 % normal build: modern PDF 1.7 (LuaTeX default is 1.5)
\fi

% Your data and settings.
\input{tfg_vars.tex}
% Template metadata (version and provenance); maintained by whoever updates the
% template, you need not touch it for your thesis.
\input{tfg_template_info.tex}

\documentclass[twoside,12pt]{book}

% UC3M TFG style: packages, layout, fonts and PDF metadata.
% Keep this as the first package; it loads everything else.
\usepackage{tfg_uc3m}

% Bibliography: style chosen by \bibStyle in tfg_vars.tex (apa / ieee).
\custombib{\bibStyle}
\addbibresource{bibliography/bibliography.bib}

% Glossary and acronym entries.
\loadglsentries{others/glossary}

\begin{document}
\selectlanguage{english}

% =====================================================================
%  FRONT MATTER
% =====================================================================
% Roman numbering from the very start, so the cover/acknowledgments and the
% body never share page anchors (avoids duplicate-destination warnings).
\pagenumbering{Roman}
\pagestyle{empty}

% --- Cover page ------------------------------------------------------
\makecover
\cleardoublepage

% --- Acknowledgments / dedication ------------------------------------
\input{others/acks}
\clearpage

\pagestyle{plain}

% --- Abstract (English) and Resumen (Spanish) ------------------------
% UC3M usually requires both summaries. Remove the one you do not need.
\input{others/abstract}
\clearpage
\input{others/resumen}
\clearpage

% --- Tables of contents, figures and tables --------------------------
% Chained with NO blank pages between them: inside the group, the
% \cleardoublepage that \tableofcontents/\listof... insert (book class,
% openright) is downgraded to \clearpage, so each list starts right after
% the previous one instead of forcing an odd page.
\begingroup
\makeatletter\@openrightfalse\makeatother
\let\cleardoublepage\clearpage
\tableofcontents
\listoffigures
\listoftables
\endgroup
\clearpage

% =====================================================================
%  BODY
% =====================================================================
\pagenumbering{arabic}              % Arabic numbering from the first chapter
\pagestyle{tfgbody}

% Demo chapter showing the main LaTeX features of this template.
% Shown only if \showDemoChapter = true (in tfg_vars.tex); set it to false for

% =====================================================================
%  BACK MATTER
% =====================================================================
% --- Acronyms and glossary -------------------------------------------
\printglossary[type=\acronymtype,title=Acronyms]
\printglossary[type=main]

% --- Bibliography ----------------------------------------------------
\printbibliography[heading=bibintoc]

% --- Annexes / appendices --------------------------------------------
\input{chapters/09_annexes}

% --- Mandatory annex: Generative AI use declaration ------------------
\input{others/ai_declaration}
\cleardoublepage

\end{document}
